-- Custom pattern-matching library: Resolves hierarchical instance trees with automated destructors

local RootAlias = "root"

local function parsePath(pathStr: string): { string }
	local pathArray = string.split(pathStr, "/")
	for idx = #pathArray, 1, -1 do
		if pathArray[idx] == "" then
			table.remove(pathArray, idx)
		end
	end
	return pathArray
end

local function isManifestResolved(resolvedManifest: { [string]: Instance }, manifestSizeTarget: number): boolean
	local manifestSize = 0
	for _ in pairs(resolvedManifest) do
		manifestSize += 1
	end
	assert(manifestSize <= manifestSizeTarget, tostring(manifestSize))
	return manifestSize == manifestSizeTarget
end

local function unbindNodeDescend(node: any, resolvedManifest: { [string]: Instance })
	if node.instance == nil then return end
	node.instance = nil
	local connections = node.connections
	if connections then
		for _, conn in ipairs(connections) do
			conn:Disconnect()
		end
		table.clear(connections)
	end
	if resolvedManifest and node.alias then
		resolvedManifest[node.alias] = nil
	end
	local children = node.children
	if children then
		for _, childNode in pairs(children) do
			unbindNodeDescend(childNode, resolvedManifest)
		end
	end
end

local AtomicBinding = {}
AtomicBinding.__index = AtomicBinding

export type Destructor = () -> () | RBXScriptConnection | Instance | { Destroy: (any) -> () } | { destroy: (any) -> () } | any
export type BoundCallback = (manifest: { [string]: Instance }) -> Destructor?

function AtomicBinding.new(manifest: { [string]: string }, boundFn: BoundCallback)
	local parsedManifest = {}
	local manifestSizeTarget = 1
	for alias, rawPath in pairs(manifest) do
		parsedManifest[alias] = parsePath(rawPath)
		manifestSizeTarget += 1
	end
	return setmetatable({
		_boundFn = boundFn,
		_parsedManifest = parsedManifest,
		_manifestSizeTarget = manifestSizeTarget,
		_dtorMap = {} :: { [Instance]: Destructor? },
		_connections = {},
		_rootInstToRootNode = {},
		_rootInstToManifest = {} :: { [Instance]: { [string]: Instance } },
	}, AtomicBinding)
end

function AtomicBinding:_startBoundFn(root: Instance, resolvedManifest: { [string]: Instance })
	if self._dtorMap[root] ~= nil then
		return
	end
	local success, dtor = pcall(self._boundFn, resolvedManifest)
	if success then
		self._dtorMap[root] = dtor
	else
		warn(string.format("[AtomicBinding] Error executing bound callback on %s: %s", root:GetFullName(), tostring(dtor)))
	end
end

function AtomicBinding:_stopBoundFn(root: Instance)
	local dtor = self._dtorMap[root]
	if dtor == nil then return end
	self._dtorMap[root] = nil

	pcall(function()
		local dtorType = typeof(dtor)
		if dtorType == "function" then
			(dtor :: () -> ())()
		elseif dtorType == "RBXScriptConnection" then
			(dtor :: RBXScriptConnection):Disconnect()
		elseif dtorType == "Instance" then
			(dtor :: Instance):Destroy()
		elseif dtorType == "table" then
			local tbl = dtor :: any
			if typeof(tbl.Destroy) == "function" then
				tbl:Destroy()
			elseif typeof(tbl.destroy) == "function" then
				tbl:destroy()
			elseif typeof(tbl.Disconnect) == "function" then
				tbl:Disconnect()
			end
		end
	end)
end

function AtomicBinding:bindRoot(root: Instance)
	debug.profilebegin("AtomicBinding:BindRoot")
	local parsedManifest = self._parsedManifest
	local rootInstToRootNode = self._rootInstToRootNode
	local rootInstToManifest = self._rootInstToManifest
	local manifestSizeTarget = self._manifestSizeTarget
	assert(rootInstToManifest[root] == nil, "Root instance is already bound")

	local resolvedManifest: { [string]: Instance } = {}
	rootInstToManifest[root] = resolvedManifest

	local rootNode = {
		alias = RootAlias,
		instance = root,
		children = {},
		connections = {},
	}
	rootInstToRootNode[root] = rootNode

	-- Build hierarchical search tree from path segments
	for alias, parsedPath in pairs(parsedManifest) do
		local parentNode = rootNode
		for idx, childName in ipairs(parsedPath) do
			local leaf = idx == #parsedPath
			local childNode = parentNode.children[childName] or {}
			if leaf then
				childNode.alias = alias
			else
				childNode.children = childNode.children or {}
				childNode.connections = childNode.connections or {}
			end
			parentNode.children[childName] = childNode
			parentNode = childNode
		end
	end

	local function processNode(node: any)
		local instance = assert(node.instance)
		local children = node.children
		if node.alias then resolvedManifest[node.alias] = instance end

		if children then
			local function processAddChild(childInstance: Instance)
				local childNode = children[childInstance.Name]
				if not childNode or childNode.instance ~= nil then return end
				childNode.instance = childInstance
				processNode(childNode)
			end

			local function processDeleteChild(childInstance: Instance)
				local childNode = children[childInstance.Name]
				if not childNode or childNode.instance ~= childInstance then return end
				self:_stopBoundFn(root)
				unbindNodeDescend(childNode, resolvedManifest)
			end

			for _, child in ipairs(instance:GetChildren()) do processAddChild(child) end
			table.insert(node.connections, instance.ChildAdded:Connect(processAddChild))
			table.insert(node.connections, instance.ChildRemoved:Connect(processDeleteChild))
		end

		if not children and isManifestResolved(resolvedManifest, manifestSizeTarget) then
			self:_startBoundFn(root, resolvedManifest)
		end
	end

	processNode(rootNode)
	debug.profileend()
end

function AtomicBinding:unbindRoot(root: Instance)
	self:_stopBoundFn(root)
	local rootNode = self._rootInstToRootNode[root]
	if rootNode then
		unbindNodeDescend(rootNode, self._rootInstToManifest[root])
		self._rootInstToRootNode[root] = nil
	end
	self._rootInstToManifest[root] = nil
end

function AtomicBinding:destroy()
	for root in pairs(self._rootInstToManifest) do
		self:unbindRoot(root)
	end
	table.clear(self._rootInstToRootNode)
	table.clear(self._rootInstToManifest)
	table.clear(self._dtorMap)
end

return AtomicBinding

local function CalculateScopeBootPercent(CompletedStepCount: number, TotalStepCount: number): number
	if TotalStepCount <= 0 then return 100 end
	local RawPercent = (CompletedStepCount / TotalStepCount) * 100
	return math.clamp(math.floor(RawPercent + 0.5), 0, 100)

local function CountLifecyclePhaseSteps(LifecycleUnits: { LifecycleUnit }, PhaseName: string): number
	local Count = 0
	for _, Unit in ipairs(LifecycleUnits) do
		local Loaded = Unit.Loaded
		if type(Loaded) == "table" and type(Loaded[PhaseName]) == "function" then
			Count += 1
		end
	end
	return Count
end

local function ExecuteLifecyclePhase(ScopeName: string, LifecycleUnits: { LifecycleUnit }, PhaseName: string, Context: any)
	for _, Unit in ipairs(LifecycleUnits) do
		local Loaded = Unit.Loaded
		if type(Loaded) == "table" and type(Loaded[PhaseName]) == "function" then
			debug.profilebegin(ScopeName .. ":" .. Unit.Name .. ":" .. PhaseName)
			local success, err = pcall(function()
				Loaded[PhaseName](Context)
			end)
			debug.profileend()
			if not success then
				error(string.format("[ModuleLoader] Error during %s on %s: %s", PhaseName, Unit.Name, tostring(err)))
			end
		end
	end
end

-- Resolve module instance from a dot-delimited path
local function ResolveModuleInstance(scope: string, path: string): ModuleScript?
	local current: Instance? = nil

	if scope == "Shared" or path:match("^Packages%.") or path:match("^SharedModules%.") then
		current = ReplicatedStorage
	elseif scope == "Server" or path:match("^ServerModules%.") then
		local sss = game:GetService("ServerScriptService")
		current = sss
	else
		current = ReplicatedStorage
	end

	local parts = string.split(path, ".")
	for _, part in ipairs(parts) do
		if not current then return nil end
		current = current:FindFirstChild(part)
	end

	if current and current:IsA("ModuleScript") then
		return current :: ModuleScript
	end
	return nil
end

function ModuleLoader.Register(name: string, module: any)
	ModuleCache[name] = module
end

function ModuleLoader.Require(scope: string, path: string): any
	local cacheKey = scope .. ":" .. path
	if ModuleCache[cacheKey] ~= nil then
		return ModuleCache[cacheKey]
	end
	if ModuleCache[path] ~= nil then
		return ModuleCache[path]
	end

	local inst = ResolveModuleInstance(scope, path)
	if inst then
		local loaded = require(inst)
		ModuleCache[cacheKey] = loaded
		ModuleCache[path] = loaded
		return loaded
	end

	-- Check fallback in cache under simple name
	local simpleName = path:match("([^%.]+)$") or path
	if ModuleCache[simpleName] ~= nil then
		return ModuleCache[simpleName]
	end

	error(string.format("[ModuleLoader] Could not resolve module '%s' in scope '%s'", path, scope))
end

function ModuleLoader.RequireShared(path: string): any
	return ModuleLoader.Require("Shared", path)
end

function ModuleLoader.RequireScoped(path: string): any
	local isServer = RunService:IsServer()
	local scope = if isServer then "Server" else "Client"
	return ModuleLoader.Require(scope, path)
end

function ModuleLoader.Boot(scopeName: string, container: Instance)
	local units: { LifecycleUnit } = {}

	for _, descendant in ipairs(container:GetDescendants()) do
		if descendant:IsA("ModuleScript") then
			local success, loaded = pcall(require, descendant)
			if success then
				local priority = 10
				if type(loaded) == "table" and type(loaded.Metadata) == "table" and type(loaded.Metadata.Priority) == "number" then
					priority = loaded.Metadata.Priority
				end
				table.insert(units, {
					Name = descendant.Name,
					Loaded = loaded,
					Priority = priority,
				})
				ModuleCache[descendant.Name] = loaded
			else
				warn(string.format("[ModuleLoader] Failed to require %s: %s", descendant:GetFullName(), tostring(loaded)))
			end
		end
	end

	-- Sort by Priority descending (higher priority inits first)
	table.sort(units, function(a, b)
		return a.Priority > b.Priority
	end)

	local context = {
		ModuleLoader = ModuleLoader,
	}

	-- Phase 1: Synchronous Init
	ExecuteLifecyclePhase(scopeName, units, "Init", context)

	-- Phase 2: Asynchronous Start
	ExecuteLifecyclePhase(scopeName, units, "Start", context)

function BufferUtil.CreateWriter(initialSize: number?)

		rayParams.FilterDescendantsInstances = excludeList
	end
	rayParams.IgnoreWater = true

	local hitsRecorded: { HitResult } = {}
	local hitVictims: { [Model]: boolean } = {}

	local sphereRadius = math.max(request.Size.X, request.Size.Y, request.Size.Z) / 2

	-- Iterative Penetration Shapecasting (supports multi-target cleave)

		end

			break
		end

-- Internal: Load Session with Session-Locking & Exponential Backoff
function DataService._loadSession(player: Player)
	-- Guard against race conditions: abort if player disconnected before thread executed
	if not player:IsDescendantOf(Players) then return end
	if _profiles[player] then return end

	local attempts = 0
	local profileData: PlayerData? = nil
	local saveToken = HttpService:GenerateGUID(false)

	while attempts < MAX_RETRY_ATTEMPTS do
		if not player:IsDescendantOf(Players) then return end
		attempts += 1
		local success, result = pcall(function()
			-- In production deployment with ProfileStore backend, this executes:
			-- return ProfileStore:LoadProfileAsync(playerKey)
			return {
				Coins = 1_250,
				Gems = 50,
				Level = 12,

				Stats = { Playtime = 3_600, ChestsOpened = 14 },
				LastSaveToken = saveToken,
				Version = DATA_SCHEMA_VERSION,
			} :: PlayerData
		end)

		if success and result then
			profileData = result
			break
		end

		-- Exponential backoff: base * 2^(attempts - 1)
		local backoff = EXP_BACKOFF_BASE * (2 ^ (attempts - 1))
		task.wait(backoff)
		-- Guard against race conditions: check if player disconnected during backoff yield
		if not player:IsDescendantOf(Players) then return end
	end

	-- Final disconnect guard before session allocation to prevent orphaned memory leaks
	if not player:IsDescendantOf(Players) then return end

	if profileData then
		_profiles[player] = profileData
		_saveTokens[player] = saveToken
		_readOnlyMode[player] = false
		print(string.format("[DataService] Session locked for %s (Token: %s)", player.Name, saveToken:sub(1, 8)))
	else
		-- Fail-safe fallback into Read-Only mode to protect user data (using isolated fresh table)
		_readOnlyMode[player] = true
		_profiles[player] = _createDefaultProfile()
		warn(string.format("[DataService] Failed to lock session for %s. Operating in READ-ONLY mode.", player.Name))
	end
end

		return false, nil

	if _readOnlyMode[player] then
		warn(string.format("[DataService] Cannot update read-only session for %s", player.Name))
		return false, nil
	end

	local rawProfile = profile :: any
	local currentValue = rawProfile[key]

	-- Execute atomic modification callback
	local success, updatedValue, shouldPersist = pcall(function()

	rawProfile[key] = updatedValue

	local nextToken = HttpService:GenerateGUID(false)

--!strict
-- [ProductionCore / Packages / Scythe.luau]
-- Enterprise Zero-Allocation Scope-Based Memory Management Engine.
-- Provides ultra-low-overhead resource lifecycle and cleanup via numeric scope handles.

export type ScopeId = number
export type CleanupTask = Instance | RBXScriptConnection | () -> () | thread | ScopeId | { Destroy: (any) -> () } | { destroy: (any) -> () } | { Disconnect: (any) -> () } | { disconnect: (any) -> () } | any

local _scopes: { [ScopeId]: { CleanupTask } } = {}
local _nextScopeId: ScopeId = 0

local function _cleanupResource(resource: CleanupTask)
	local resType = typeof(resource)

	if resType == "RBXScriptConnection" then
		(resource :: RBXScriptConnection):Disconnect()
	elseif resType == "Instance" then
		(resource :: Instance):Destroy()
	elseif resType == "function" then
		(resource :: () -> ())()
	elseif resType == "thread" then
		task.cancel(resource :: thread)
	elseif resType == "number" then
		local childScopeId = resource :: ScopeId
		if _scopes[childScopeId] ~= nil then
			local bucket = _scopes[childScopeId]
			_scopes[childScopeId] = nil
			for i = #bucket, 1, -1 do
				local ok, err = pcall(_cleanupResource, bucket[i])
				if not ok then
					warn(string.format("[Scythe] Cleanup error in nested scope %d: %s", childScopeId, tostring(err)))
				end
			end
		end
	elseif resType == "table" then
		local tbl = resource :: any
		if typeof(tbl.Destroy) == "function" then
			tbl:Destroy()
		elseif typeof(tbl.destroy) == "function" then
			tbl:destroy()
		elseif typeof(tbl.Disconnect) == "function" then
			tbl:Disconnect()
		elseif typeof(tbl.disconnect) == "function" then
			tbl:disconnect()
		end
	end
end

local Scythe = {}

-- Create a new scope handle or destroy an existing scope
function Scythe.scope(scopeId: ScopeId?): ScopeId?
	if scopeId == nil then
		_nextScopeId += 1
		local id = _nextScopeId
		_scopes[id] = {}
		return id
	end

	local bucket = _scopes[scopeId]
	if not bucket then
		return nil
	end

	_scopes[scopeId] = nil

	-- Clean up resources in reverse order (LIFO)
	for i = #bucket, 1, -1 do
		local taskItem = bucket[i]
		local ok, err = pcall(_cleanupResource, taskItem)
		if not ok then
			warn(string.format("[Scythe] Cleanup error in scope %d: %s", scopeId, tostring(err)))
		end
	end

	return nil
end

-- Register a resource inside the specified scope
function Scythe.add(scopeId: ScopeId, resource: CleanupTask): CleanupTask
	local bucket = _scopes[scopeId]
	if not bucket then
		warn(string.format("[Scythe] Attempted to add resource to dead or invalid scope %d. Cleaning immediately.", scopeId))
		pcall(_cleanupResource, resource)
		return resource
	end

	table.insert(bucket, resource)
	return resource
end

-- Batch register an array of resources inside the specified scope
function Scythe.addBulk(scopeId: ScopeId, resources: { CleanupTask })
	local bucket = _scopes[scopeId]
	if not bucket then
		warn(string.format("[Scythe] Attempted to addBulk resources to dead or invalid scope %d. Cleaning immediately.", scopeId))
		for _, res in resources do
			pcall(_cleanupResource, res)
		end
		return
	end

	for _, res in resources do
		table.insert(bucket, res)
	end
end

-- Remove a resource from an active scope without destroying it immediately
function Scythe.remove(scopeId: ScopeId, resource: CleanupTask): boolean
	local bucket = _scopes[scopeId]
	if not bucket then return false end

	local idx = table.find(bucket, resource)
	if idx then
		table.remove(bucket, idx)
		return true
	end
	return false
end

-- Check if the scope is currently active
function Scythe.isAlive(scopeId: ScopeId): boolean
	return _scopes[scopeId] ~= nil
end

return table.freeze(Scythe)
